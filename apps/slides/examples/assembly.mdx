import { Text, View } from "react-native"
import { Journey, Camera, Choreography, Floor, Bloom, Fog, Vignette, Grain, PointLight, Material, primitive, cardInstances } from "slides:journey/Journey"

export const colors = ["#79dbff", "#a7b7ff", "#efb96b"]
export const mesh = (kind, scale, color) => [primitive(kind, { size: 0.5, scale, color, emissive: 0.12, material: "alloy" })]
export const grid = Array.from({ length: 18 }, (_, i) => ({ id: `tile-${i}`, home: {
  position: [(i % 6 - 2.5) * 1.72, (1 - Math.floor(i / 6)) * 1.15 + 0.8, 0],
  rotation: [0, 0, 0], color: colors[Math.floor(i / 6)] }, instances: mesh("box", [1.4, 0.8, 0.45], colors[Math.floor(i / 6)]) }))
export const arrival = grid.map((part, i) => ({ ...part, home: { ...part.home,
  position: [(i % 2 ? 1 : -1) * (12 + i * 0.15), part.home.position[1] + (i % 3 - 1) * 2, -5 - i * 0.2],
  rotation: [0, (i % 2 ? 1 : -1) * 0.8, 0] } }))
export const nodes = Array.from({ length: 7 }, (_, i) => ({ id: `node-${i}`, home: {
  position: i === 0 ? [0, 3, 0] : i < 3 ? [(i === 1 ? -1 : 1) * 3, 1, 0] : [(i - 4.5) * 2.4, -1.2, 0],
  color: colors[i % 3] }, instances: cardInstances({ scale: [1.65, 0.74, 0.2], radius: 0.12, color: colors[i % 3], emissive: 0.09, avatar: 0 }) }))
export const bases = Array.from({ length: 3 }, (_, i) => ({ id: `base-${i}`, home: {
  position: [(i - 1) * 4.5, -3 + 0.65 / 2, 0], rotation: [0, 0.12 * (i - 1), 0], color: colors[i] },
  instances: mesh("box", [3.2, 0.65, 2.2], colors[i]) }))
export const layouts = [{ id: "arrival", parts: arrival }, { id: "tiles", parts: grid }, { id: "nodes", parts: nodes }, { id: "bases", parts: bases }]
export const transitions = [
  { from: "arrival", to: "tiles", at: 0, duration: 1.65, stagger: 0.045, overshoot: 0.1, swapAt: 0.55,
    onLand: { count: 16, seed: 13, color: "#ffd499", speed: 4.5, lifetime: 0.6 },
    mapping: grid.map((part, i) => ({ from: part.id, to: part.id, via: [
      [arrival[i].home.position[0], grid[i].home.position[1], arrival[i].home.position[2]],
      [grid[i].home.position[0], grid[i].home.position[1], -2.5]] })) },
  { from: "tiles", to: "nodes", step: 1, duration: 2.1, stagger: 0.045, swapAt: 0.58, overshoot: 0.09,
    onLand: { count: 16, seed: 29, speed: 4.2 } },
  { from: "nodes", to: "bases", step: 2, duration: 1.8, stagger: 0.075, swapAt: 0.52, overshoot: 0.07,
    onLand: { count: 24, seed: 47, color: "#fff0b7", speed: 5.2 } },
]

<Background>
  <Journey background quality="high" renderScale={1} exposure={1.15} previewTime={4}>
    <Camera shots={[
      { step: 0, position: [8, 5, 19], target: [0, 0.25, 0], fov: 38 },
      { step: 1, position: [4, 3.5, 20], target: [0, 0.25, 0], duration: 2.5, ease: "smoothstep" },
      { step: 2, position: [7, 5.5, 20], target: [0, -1.5, 0], duration: 2.4, ease: "smoothstep" },
    ]} />
    <Material id="alloy" roughness={0.3} metal={0.68} rim={0.12} />
    <PointLight position={[-5, 10, 9]} color="#b8dcff" intensity={125} range={45} />
    <PointLight position={[8, 4, -5]} color="#f2bd79" intensity={90} range={32} />
    <Choreography id="assembly" layouts={layouts} transitions={transitions} />
    <Floor y={-3} size={70} color="#101c2b" roughness={0.16} />
    <Fog color="#060b15" density={0.009} />
    <Bloom intensity={0.65} threshold={1.4} radius={0.55} />
    <Vignette amount={0.24} /><Grain amount={0.006} />
  </Journey>
</Background>

<Steps count={3}>{step => <View style={{ position: "absolute", top: 16, left: 0, right: 0 }}>
  <Text style={{ color: "#c5d7eb", fontFamily: "Menlo", fontSize: 22, letterSpacing: 4 }}>{["18 PARTS", "7 NODES", "3 BASES"][step]}</Text>
</View>}</Steps>
