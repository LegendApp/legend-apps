import { Text, View } from "react-native"
import { RenderTree } from "slides:charts/RenderTree"
import { AnimatedNumber, NumberCard } from "slides:charts/AnimatedNumber"
import { BarChart } from "slides:charts/BarChart"
import { LineChart } from "slides:charts/LineChart"
import { World, Camera, Material, RenderTree as WorldRenderTree, RimLight, Bloom, Fog, Vignette, Grain, Label, tree } from "slides:world/World"

export const nodes = [
  { id: "app", label: "App" },
  { id: "header", label: "Header" }, { id: "feed", label: "Feed" }, { id: "side", label: "Sidebar" },
  { id: "logo", label: "Logo" }, { id: "search", label: "Search" },
  { id: "a", label: "Item A" }, { id: "b", label: "Item B" }, { id: "c", label: "Item C" },
  { id: "profile", label: "Profile" },
]
export const edges = [["app", "header"], ["app", "feed"], ["app", "side"], ["header", "logo"], ["header", "search"],
  ["feed", "a"], ["feed", "b"], ["feed", "c"], ["side", "profile"]]
export const flashes = [undefined, "all", { subtree: "feed" }, ["b"], "all"]
export const modes = ["", "all", "subtree", "single leaf", "all"]

export const mode = text => <Text style={{ color: "#7b8ba1", fontFamily: "Menlo", fontSize: 24, letterSpacing: 2.4, height: 34 }}>{text.toUpperCase()}</Text>

# Render tree

<Steps count={5}>{step => <View style={{ gap: 20 }}>
  {mode(modes[step])}
  <RenderTree nodes={nodes} edges={edges} flashes={flashes} step={step} width={1680} height={560} />
</View>}</Steps>

<!-- Step 1 renders every node, step 2 one subtree, step 3 a single leaf, step 4 everything again. Badges and the total accumulate. -->

# Time series

export const wave = (seed, spikes) => Array.from({ length: 48 }, (_, i) => {
  const base = 9 + 2.2 * Math.sin(i * 0.7 + seed) + 1.4 * Math.sin(i * 1.9 + seed * 2)
  return spikes && (i * 7 + 3) % 11 === 0 ? base + 26 + ((i * 13) % 9) * 2.2 : base
})
export const frames = [wave(0, true), wave(0, false)]

<Steps count={2}>{step => <LineChart series={[{ points: frames[step] }]} maximum={60} width={1680} height={600}
  reference={{ value: 16.7, label: "16.7 ms" }} xLabels={["0 s", "1 s", "2 s", "3 s", "4 s"]} />}</Steps>

<!-- Step 1 eases every point to the flattened dataset. -->
